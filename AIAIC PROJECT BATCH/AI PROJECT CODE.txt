<script>
    // --- Role Switch Logic ---
    let currentRole = "student"; // Track current role

    const studentRoleBtn = document.getElementById('studentRoleBtn');
    const facultyRoleBtn = document.getElementById('facultyRoleBtn');
    const studentLoginBox = document.getElementById('studentLoginBox');
    const facultyLoginBox = document.getElementById('facultyLoginBox');

    studentRoleBtn.onclick = function() {
        currentRole = "student";
        studentRoleBtn.classList.add('active');
        facultyRoleBtn.classList.remove('active');
        studentLoginBox.style.display = 'block';
        facultyLoginBox.style.display = 'none';
    };
    facultyRoleBtn.onclick = function() {
        currentRole = "faculty";
        facultyRoleBtn.classList.add('active');
        studentRoleBtn.classList.remove('active');
        studentLoginBox.style.display = 'none';
        facultyLoginBox.style.display = 'block';
    };

    // --- Student Login/Signup Logic ---
    const loginScreen = document.getElementById('loginScreen');
    const splashScreen = document.getElementById('splashScreen');
    const mainContent = document.getElementById('mainContent');
    const loginBox = document.getElementById('loginBox');
    const signupBox = document.getElementById('signupBox');
    const loginBtn = document.getElementById('loginBtn');
    const signupBtn = document.getElementById('signupBtn');
    const showSignupBtn = document.getElementById('showSignupBtn');
    const showLoginBtn = document.getElementById('showLoginBtn');
    const enrolmentInput = document.getElementById('enrolment');
    const passwordInput = document.getElementById('password');
    const loginError = document.getElementById('loginError');
    const signupError = document.getElementById('signupError');
    const signupSuccess = document.getElementById('signupSuccess');
    const signupEnrolment = document.getElementById('signupEnrolment');
    const signupPassword = document.getElementById('signupPassword');

    function getStudents() {
        return JSON.parse(localStorage.getItem("students") || "[]");
    }
    function setStudents(students) {
        localStorage.setItem("students", JSON.stringify(students));
    }
    if (!localStorage.getItem("students")) {
        setStudents([
            { enrolment: "student1", password: "pass123" },
            { enrolment: "student2", password: "abc456" },
            { enrolment: "student3", password: "qwerty" }
        ]);
    }
    function checkLogin(enrolment, password) {
        return getStudents().some(
            s => s.enrolment === enrolment && s.password === password
        );
    }
    loginBtn.onclick = function() {
        const enrolment = enrolmentInput.value.trim();
        const password = passwordInput.value;
        if (checkLogin(enrolment, password)) {
            currentRole = "student";
            loginScreen.style.opacity = 0;
            setTimeout(() => {
                loginScreen.style.display = "none";
                mainContent.style.display = "block";
                // Initialize batch filter for students
                if (batchFilter) {
                    populateBatchFilter();
                    const savedBatch = localStorage.getItem('selectedBatch');
                    if (savedBatch) {
                        batchFilter.value = savedBatch;
                    } else {
                        batchFilter.value = 'All';
                    }
                }
                // Make batch filter visible for students
                const batchFilterLabel = document.getElementById("batchFilterLabel");
                if (batchFilterLabel) {
                    batchFilterLabel.style.display = "block";
                }
                renderRooms();
                updateBookingSection();
                startMainApp();
            }, 400);
        } else {
            loginError.textContent = "Invalid enrolment or password.";
            loginError.style.display = "block";
        }
    };
    [enrolmentInput, passwordInput].forEach(input => {
        input.addEventListener("keydown", function(e) {
            if (e.key === "Enter") loginBtn.click();
        });
    });
    showSignupBtn.onclick = function() {
        loginBox.style.display = "none";
        signupBox.style.display = "flex";
        loginError.style.display = "none";
        signupError.style.display = "none";
        signupSuccess.style.display = "none";
        signupEnrolment.value = "";
        signupPassword.value = "";
    };
    showLoginBtn.onclick = function() {
        signupBox.style.display = "none";
        loginBox.style.display = "flex";
        loginError.style.display = "none";
        signupError.style.display = "none";
        signupSuccess.style.display = "none";
    };
    signupBtn.onclick = function() {
        const enrolment = signupEnrolment.value.trim();
        const password = signupPassword.value;
        if (!enrolment || !password) {
            signupError.textContent = "Please fill all fields.";
            signupError.style.display = "block";
            signupSuccess.style.display = "none";
            return;
        }
        let students = getStudents();
        if (students.some(s => s.enrolment === enrolment)) {
            signupError.textContent = "Enrolment already exists.";
            signupError.style.display = "block";
            signupSuccess.style.display = "none";
            return;
        }
        students.push({ enrolment, password });
        setStudents(students);
        signupError.style.display = "none";
        signupSuccess.textContent = "Sign up successful! You can now log in.";
        signupSuccess.style.display = "block";
        setTimeout(() => {
            signupBox.style.display = "none";
            loginBox.style.display = "flex";
        }, 1200);
    };

    function formatTimeLeft(ms) {
        if (ms <= 0) return "Expired";
        const minutes = Math.floor(ms / 60000);
        const seconds = Math.floor((ms % 60000) / 1000);
        return `${minutes}:${seconds.toString().padStart(2, "0")}`;
    }

    // Format date from YYYY-MM-DD to readable format (e.g., "12 Nov 2025")
    function formatDate(dateStr) {
        if (!dateStr) return "N/A";
        const date = new Date(dateStr + 'T00:00:00');
        const months = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
        return `${date.getDate()} ${months[date.getMonth()]} ${date.getFullYear()}`;
    }

    // Format time from HH:MM to readable format (e.g., "10 AM" or "2 PM")
    function formatTime(timeStr) {
        if (!timeStr) return "N/A";
        const [hours, minutes] = timeStr.split(':');
        const hour = parseInt(hours);
        const ampm = hour >= 12 ? 'PM' : 'AM';
        const hour12 = hour % 12 || 12;
        return `${hour12}:${minutes} ${ampm}`;
    }

    // Format time range (e.g., "10 AM - 12 PM")
    function formatTimeRange(startTime, endTime) {
        return `${formatTime(startTime)} - ${formatTime(endTime)}`;
    }

    function updateRoomsAvailabilityFromBookings() {
        const now = Date.now();
        
        rooms.forEach(room => {
            const bookingData = bookings[room.number];
            
            if (!bookingData) {
                // No booking for this room - it's free
                room.available = true;
            } else {
                // Check if booking has expired
                const endTime = typeof bookingData === 'number' ? bookingData : bookingData?.endTime;
                
                if (endTime && endTime > now) {
                    // Booking is still active - room is occupied
                    room.available = false;
                } else {
                    // Booking has expired - room is free
                    room.available = true;
                }
            }
        });
    }

    // Update the batch status message shown to students
    // This function displays ALL classes booked by faculty for the selected batch number
    function updateBatchStatus() {
        const batchStatus = document.getElementById('batchStatus');
        if (!batchStatus || !batchFilter || currentRole !== 'student') return;
        const sel = batchFilter.value || 'All';
        if (sel === 'All') {
            batchStatus.innerHTML = '<strong style="color:#4f8cff;">Please select a batch number to view classes</strong>';
            return;
        }
        const now = Date.now();
        const selStr = String(sel); // Ensure string comparison
        const roomsForBatch = [];
        // Scan ALL bookings to find classes for the selected batch
        for (const [roomNum, bd] of Object.entries(bookings)) {
            if (!bd || typeof bd !== 'object') continue;
            // Only show bookings made by faculty
            if (bd.bookedBy !== 'faculty') continue;
            
            // Check if batch numbers array exists and includes the selected batch
            if (!Array.isArray(bd.batchNumbers) || bd.batchNumbers.length === 0) continue;
            
            // Convert all batch numbers to strings for comparison
            const batchNumbersStr = bd.batchNumbers.map(b => String(b));
            if (!batchNumbersStr.includes(selStr)) continue;
            
            const endTime = bd.endTime;
            if (!endTime) continue;
            
            // Show bookings that haven't expired (including future bookings)
            // Use same logic as renderRooms - show if not expired more than 1 hour ago
            const oneHourInMs = 60 * 60 * 1000;
            if (endTime < (now - oneHourInMs)) continue; // expired more than 1 hour ago
            
            // Find the room details
            const room = rooms.find(r => r.number === roomNum);
            const date = formatDate(bd.bookingDate);
            const timeRange = bd.startTime && bd.endTimeDisplay ? formatTimeRange(bd.startTime, bd.endTimeDisplay) : 'N/A';
            const faculty = bd.facultyName || 'Unknown Faculty';
            
            roomsForBatch.push({ 
                room: roomNum, 
                block: room ? room.block : 'N/A',
                type: room ? room.type : 'N/A',
                subject: bd.subjectName || 'N/A',
                date: date,
                time: timeRange,
                faculty: faculty,
                until: formatTimeLeft(endTime - now) 
            });
        }
        if (roomsForBatch.length === 0) {
            batchStatus.innerHTML = `<strong style="color:#e74c3c;">No classes are booked for batch ${sel}</strong>`;
        } else {
            // Display all classes with full details
            const list = roomsForBatch.map(r => 
                `<strong>${r.room}</strong> - ${r.subject}<br>
                📅 ${r.date} | ⏰ ${r.time} | 👨‍🏫 ${r.faculty}`
            ).join('<br><br>');
            batchStatus.innerHTML = `<strong style="color:#27ae60;">All Classes for Batch ${sel} (${roomsForBatch.length} total):</strong><br><br>${list}`;
        }
    }

        const blockFilterLabel = document.getElementById("blockFilterLabel");
        const typeFilterLabel = document.getElementById("typeFilterLabel");
        const batchFilterLabel = document.getElementById("batchFilterLabel");
        if (currentRole === "faculty") {
            facultyBookingFields.style.display = "block";
            if (blockFilterLabel) blockFilterLabel.style.display = "block";
            if (typeFilterLabel) typeFilterLabel.style.display = "block";
            if (batchFilterLabel) batchFilterLabel.style.display = "none"; // faculty doesn't need student batch selector
        } else {
            facultyBookingFields.style.display = "none";
            if (blockFilterLabel) blockFilterLabel.style.display = "none";
            if (typeFilterLabel) typeFilterLabel.style.display = "none";
            if (batchFilterLabel) batchFilterLabel.style.display = "block"; // show batch selector to students
        }
        // Availability panel removed from UI

        if (currentRole === "student") {
            bookingSection.style.display = "none";
            timerDisplay.textContent = "";
            clearInterval(timerInterval);
            return;
        }

        if (selectedRoomNumber) {
            const room = rooms.find(r => r.number === selectedRoomNumber);
            if (room && room.available) {
                bookingSection.style.display = "block";
                selectedRoomDisplay.textContent = `${room.number} (${room.type}, ${room.block})`;
                bookRoomBtn.disabled = false;
                bookingDurationInput.disabled = false;
                timerDisplay.textContent = "";
                
                // Set default date to today for faculty bookings
                if (currentRole === "faculty") {
                    const today = new Date().toISOString().split('T')[0];
                    const dateInput = document.getElementById("bookingDateInput");
                    if (dateInput && !dateInput.value) {
                        dateInput.value = today;
                    }
                }

                clearInterval(timerInterval);
            } else if (room && !room.available) {
                bookingSection.style.display = "block";
                selectedRoomDisplay.textContent = `${room.number} (${room.type}, ${room.block})`;
                bookRoomBtn.disabled = true;
                bookingDurationInput.disabled = true;
                startCountdown(room.number);
            } else {
                bookingSection.style.display = "none";
                selectedRoomNumber = null;
                localStorage.removeItem("selectedRoom");
            }
        } else {
            bookingSection.style.display = "none";
            timerDisplay.textContent = "";
            clearInterval(timerInterval);
        }
    }

    function startCountdown(roomNumber) {
        clearInterval(timerInterval);

        const bookingData = bookings[roomNumber];
        const endTime = typeof bookingData === 'number' ? bookingData : bookingData?.endTime;
        if (!endTime) {
            timerDisplay.textContent = "";
            return;
        }

        function updateTimer() {
            const now = Date.now();
            const diff = endTime - now;
            if (diff <= 0) {
                timerDisplay.textContent = "Booking time expired. Room is now free.";
                const room = rooms.find(r => r.number === roomNumber);
                if (room) room.available = true;
                delete bookings[roomNumber];
                localStorage.setItem("bookings", JSON.stringify(bookings));
                selectedRoomNumber = null;
                localStorage.removeItem("selectedRoom");
                renderRooms();
                updateBookingSection();
                clearInterval(timerInterval);
            } else {
                timerDisplay.textContent = `Time left: ${formatTimeLeft(diff)}`;
            }
        }

        updateTimer();
        timerInterval = setInterval(updateTimer, 1000);
    }

    function startMainApp() {
        // Initialize batch filter when app starts
        if (currentRole === "student" && batchFilter) {
            populateBatchFilter();
            // Restore previously selected batch if any
            const savedBatch = localStorage.getItem('selectedBatch');
            if (savedBatch) {
                batchFilter.value = savedBatch;
            } else {
                batchFilter.value = 'All';
            }
            // Ensure batch filter is visible for students
            const batchFilterLabel = document.getElementById("batchFilterLabel");
            if (batchFilterLabel) {
                batchFilterLabel.style.display = "block";
            }
        }
        renderRooms();
        updateBookingSection();

        blockFilter.addEventListener("change", () => {
            renderRooms();
            updateBookingSection();
        });
        typeFilter.addEventListener("change", () => {
            renderRooms();
            updateBookingSection();
        });
        searchInput.addEventListener("input", () => {
            renderRooms();
            updateBookingSection();
        });
        bookRoomBtn.addEventListener("click", () => {
            if (selectedRoomNumber) {
                const duration = parseInt(bookingDurationInput.value, 10);
                if (duration && duration > 0) {
                    // Validate faculty fields if faculty is logged in
                    if (currentRole === "faculty") {
                        const batchNumbers = Array.from(document.getElementById("batchNumberSelect").selectedOptions).map(option => option.value);
                        const subjectName = document.getElementById("subjectNameInput").value.trim();
                        const bookingDate = document.getElementById("bookingDateInput").value;
                        const startTime = document.getElementById("startTimeInput").value;
                        const endTime = document.getElementById("endTimeInput").value;
                        
                        if (batchNumbers.length === 0) {
                            alert("Please select at least one batch number.");
                            return;
                        }
                        
                        if (!subjectName) {
                            alert("Please enter a subject name.");
                            return;
                        }
                        
                        if (!bookingDate) {
                            alert("Please select a date.");
                            return;
                        }
                        
                        if (!startTime || !endTime) {
                            alert("Please select both start and end times.");
                            return;
                        }
                        
                        // Calculate endTime timestamp for timer functionality
                        const dateTimeStr = `${bookingDate}T${endTime}`;
                        const endTimeTimestamp = new Date(dateTimeStr).getTime();
                        const now = Date.now();
                        
                        // Store faculty booking data with all details
                        // Ensure batch numbers are stored as strings for consistent comparison
                        const batchNumbersStr = batchNumbers.map(b => String(b));
                        bookings[selectedRoomNumber] = {
                            endTime: endTimeTimestamp,
                            batchNumbers: batchNumbersStr,
                            subjectName: subjectName,
                            bookedBy: "faculty",
                            facultyName: currentFacultyId || "Unknown Faculty",
                            bookingDate: bookingDate,
                            startTime: startTime,
                            endTimeDisplay: endTime
                        };
                    } else {
                        // Student booking (simple format for backward compatibility)
                        const now = Date.now();
                        const endTime = now + duration * 60000;
                        bookings[selectedRoomNumber] = endTime;
                    }
                    
                    localStorage.setItem("bookings", JSON.stringify(bookings));
                    renderRooms();
                    updateBookingSection();
                    
                    // Show success message for faculty
                    if (currentRole === "faculty") {
                        alert("Room booked successfully!");
                    }
                }
            }
        });

        setInterval(() => {
            renderRooms();
        }, 1000);
    }
